import platform
import psutil

print("=== SYSTEM HEALTH CHECK ===")

# OS info
print("Operating System:", platform.system(), platform.release())

# CPU usage
print("CPU Usage:", psutil.cpu_percent(interval=1), "%")

# RAM usage
ram = psutil.virtual_memory()
print("RAM Usage:", ram.percent, "%")

# Disk usage
disk = psutil.disk_usage('/')
print("Disk Usage:", disk.percent, "%")


import socket
import subprocess

print("\n=== NETWORK CHECK ===")

# Check DNS resolution
try:
    socket.gethostbyname("www.google.com")
    print("DNS Resolution: OK")
except socket.error:
    print("DNS Resolution: FAILED")

# Ping default gateway
gateway = "192.168.1.1"  # You can change this if needed
ping_gateway = subprocess.run(["ping", "-n", "1", gateway], capture_output=True)

if ping_gateway.returncode == 0:
    print("Gateway Reachable:", gateway)
else:
    print("Gateway Unreachable:", gateway)

# Ping external site
ping_google = subprocess.run(["ping", "-n", "1", "8.8.8.8"], capture_output=True)

if ping_google.returncode == 0:
    print("Internet Connectivity: OK")
else:
    print("Internet Connectivity: FAILED")
